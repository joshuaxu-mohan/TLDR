# TLDR schedule on the Oracle VM (times are UTC; the VM runs on UTC).
# This file is the crontab of record. Install it with: crontab ~/Daily/deploy/crontab.txt

# Pipeline: 07:00 covers the overnight window, then four daytime runs
0 7 * * * cd /home/ubuntu/Daily && /home/ubuntu/Daily/venv/bin/python -m src.main --run-now --since 12h >> /home/ubuntu/Daily/logs/cron.log 2>&1
0 10,13,16,20 * * * cd /home/ubuntu/Daily && /home/ubuntu/Daily/venv/bin/python -m src.main --run-now --since 4h >> /home/ubuntu/Daily/logs/cron.log 2>&1

# Staleness alert to Telegram when nothing new has been ingested for 36 hours.
# Every 6 hours rather than hourly: each check that finds a stall sends a message.
30 */6 * * * cd /home/ubuntu/Daily && /home/ubuntu/Daily/venv/bin/python -m src.main --health-check >> /home/ubuntu/Daily/logs/cron.log 2>&1

# Nightly DB backup (verified, gzipped, 14 days kept in ~/backups)
15 3 * * * /bin/bash /home/ubuntu/Daily/deploy/backup.sh >> /home/ubuntu/Daily/logs/backup.log 2>&1
